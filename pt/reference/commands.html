Referência de Comandos FWH

O FiveWin estende a linguagem de comandos Harbour/xBase com um rico conjunto de comandos de pré-processador definidos em FiveWin.ch. Esses comandos são traduzidos em chamadas de métodos de classe em tempo de compilação, fornecendo uma sintaxe familiar no estilo xBase para criar aplicações GUI do Windows.

Comandos de Janela

DEFINE WINDOW

DEFINE WINDOW oWnd ;
   [ FROM nTop, nLeft TO nBottom, nRight ] ;
   [ TITLE cTitle ] ;
   [ STYLE nStyle ] ;
   [ MENU oMenu ] ;
   [ BRUSH oBrush ] ;
   [ ICON oIcon ] ;
   [ ICONIZE | MAXIMIZE | RESTORE ] ;
   [ COLOR cClrFore [, cClrBack ] ] ;
   [ PIXEL ] ;
   [ BORDER NONE | SINGLE ] ;
   [ NOSYSMENU | NOCAPTION ] ;
   [ NOZOOM | NOMAXIMIZE ]

ACTIVATE WINDOW

ACTIVATE WINDOW oWnd ;
   [ ICONIZED | MAXIMIZED | NORMAL ] ;
   [ ON INIT bInit ] ;
   [ ON UP bUp ] ;
   [ ON DOWN bDown ] ;
   [ ON LEFT bLeft ] ;
   [ ON RIGHT bRight ] ;
   [ ON PAINT bPaint ] ;
   [ ON RESIZE bResize ] ;
   [ ON CLICK bClick ] ;
   [ ON DBLCLICK bDblClick ] ;
   [ VALID bValid ]

Comandos de Menu

DEFINE MENU / MENUITEM

MENU oMenu
   MENUITEM [ oMi ] [ PROMPT ] cPrompt ;
      [ ACTION bAction ] ;
      [ BLOCK bAction ] ;
      [ MESSAGE cMsg ] ;
      [ ACCELERATOR nAccel ] ;
      [ WHEN bWhen ] ;
      [ CHECK ] ;
      [ CHECKED ]

   MENUITEM "File"
   MENU
      MENUITEM "&Open"   ACTION OpenFile()   ACCELERATOR ACC_CTRL_O
      MENUITEM "&Save"   ACTION SaveFile()
      SEPARATOR
      MENUITEM "E&xit"   ACTION oWnd:End()
   ENDMENU

   MENUITEM "Edit"
   MENU
      MENUITEM "&Copy"   ACTION CopyData()
      MENUITEM "&Paste"  ACTION PasteData()
   ENDMENU
ENDMENU

DEFINE POPUP

DEFINE MENU oPopup POPUP
   MENUITEM "Option 1" ACTION Action1()
   MENUITEM "Option 2" ACTION Action2()
   SEPARATOR
   MENUITEM "Option 3" ACTION Action3()
ENDMENU

ACTIVATE POPUP oPopup AT nRow, nCol OF oWnd
