TWebServer / TSocket

Fonte: source/classes/twebserv.prg (TWebServer), source/classes/tsocket.prg (TSocket)

TWebServer é o servidor HTTP integrado do FiveWin que pode servir páginas HTML, processar solicitações GET/POST e mesclar conteúdo dinâmico em templates HTML. Ele é construído sobre o TSocket, que fornece comunicação de sockets TCP/IP de baixo nível usando a API Winsock do Windows.

Exemplo: Servidor HTTP Básico

#include "FiveWin.ch"

function Main()

   local oWnd, oWS

   DEFINE WINDOW oWnd TITLE "FiveWin Web Server"

   // Criar servidor web na porta 8080
   oWS := TWebServer():New( 8080 )
   oWS:cDefPath := "www"
   oWS:cDefPage := "index.html"
   oWS:Activate()

   // Abrir navegador
   oWS:Explore( "index.html" )

   ACTIVATE WINDOW oWnd ;
      VALID ( oWS:End(), .T. )

return nil

Exemplo: Conteúdo Dinâmico

#include "FiveWin.ch"

function Main()

   local oWnd, oWS

   DEFINE WINDOW oWnd TITLE "Dynamic Web Server"

   oWS := TWebServer():New( 8080 )
   oWS:lDebug := .T.
   oWS:cLogFile := "webserver.log"
   oWS:Activate()

   oWS:Explore()

   ACTIVATE WINDOW oWnd VALID ( oWS:End(), .T. )

return nil

Exemplo: Cliente de Socket TCP Bruto

#include "FiveWin.ch"

function Main()

   local oSocket, cResponse

   oSocket := TSocket():New( 0 )

   // Conectar a um servidor remoto
   oSocket:Connect( "93.184.216.34", 80 )  // example.com

   // Enviar solicitação HTTP
   oSocket:SendData( ;
      "GET / HTTP/1.1" + CRLF + ;
      "Host: example.com" + CRLF + ;
      "Connection: close" + CRLF + CRLF )

   // Ler resposta
   oSocket:bRead := { |oSock|
      cResponse := oSock:GetData()
      ? cResponse
   }

   oSocket:bClose := { |oSock|
      ? "Conexão fechada"
      oSock:End()
   }

   // Processar eventos (em um app real, o loop de mensagens trata isso)
   SysRefresh()
   Inkey( 5 )

   oSocket:End()

return nil

Exemplo: Servidor de Chat Simples

#include "FiveWin.ch"

static aClients := {}

function Main()

   local oWnd, oSocket

   DEFINE WINDOW oWnd TITLE "Chat Server - Port 9000"

   oSocket := TSocket():New( 9000 )

   oSocket:bAccept := { ||
      local oClient := TSocket():Accept( oSocket:nSocket )
      AAdd( aClients, oClient )

      oClient:bRead := { |oSock|
         local cMsg := oSock:GetData()
         // Transmitir para todos os clientes
         AEval( aClients, { |o| o:SendData( cMsg ) } )
      }

      oClient:bClose := { |oSock|
         local nAt := AScan( aClients, { |o| o:nSocket == oSock:nSocket } )
         if nAt > 0
            ADel( aClients, nAt, .T. )
         endif
         oSock:End()
      }
   }

   oSocket:Listen()

   ACTIVATE WINDOW oWnd VALID ( oSocket:End(), .T. )

return nil

Notas

  • TWebServer cria um diretório www automaticamente se ele não existir, com uma página HTML padrão.
  • O método HtmlMerge() processa tags <%=expressão%> em arquivos HTML, avaliando expressões Harbour e incorporando os resultados.
  • Os dados de cookies das solicitações HTTP são analisados automaticamente e armazenados em oWS:cCookie.
  • TSocket usa eventos de socket assíncrono do Windows (FD_READ, FD_WRITE, FD_ACCEPT, FD_CLOSE) via loop de mensagens, portanto funciona perfeitamente com a arquitetura orientada a eventos do FiveWin.
  • Para uso em produção, considere usar um proxy reverso (nginx, Apache) na frente do TWebServer para terminação SSL e servir arquivos estáticos.
