TWebView2

Fonte: source/classes/twebview2.prg

TWebView2 incorpora o motor do navegador Microsoft Edge (Chromium) em aplicações FiveWin. Ele fornece um motor de renderização moderno HTML5/CSS3/JavaScript que pode navegar URLs, exibir HTML local, executar JavaScript a partir do Harbour e receber callbacks de JavaScript de volta ao código Harbour. Isso substitui a antiga abordagem baseada no Internet Explorer TActiveX("Shell.Explorer.2").

Comunicação Harbour <-> JavaScript

Exemplo: Navegar para uma URL

#include "FiveWin.ch"

function Main()

   local oWV

   oWV := TWebView2():New()
   oWV:SetTitle( "FiveWin Browser" )
   oWV:Navigate( "https://www.fivetechsoft.com" )
   oWV:Center()
   oWV:Run()
   oWV:End()

return nil

Exemplo: Exibir HTML Local

#include "FiveWin.ch"

function Main()

   local oWV, cHtml

   TEXT INTO cHtml
   <!DOCTYPE html>
   <html>
   <head>
      <style>
         body { font-family: Segoe UI; margin: 40px; background: #1a1a2e; color: #eee; }
         h1 { color: #e94560; }
         .card { background: #16213e; padding: 20px; border-radius: 10px; margin: 20px 0; }
         button { background: #e94560; color: white; border: none; padding: 10px 20px;
                  border-radius: 5px; cursor: pointer; font-size: 14px; }
         button:hover { background: #c73a52; }
      </style>
   </head>
   <body>
      <h1>FiveWin + WebView2</h1>
      <div class="card">
         <p>Este HTML é renderizado pelo motor Chromium dentro de uma janela FiveWin.</p>
         <button onclick="SendToFWH({action:'hello', msg:'Button clicked!'})">
            Clique Aqui
         </button>
      </div>
   </body>
   </html>
   ENDTEXT

   oWV := TWebView2():New()
   oWV:SetTitle( "HTML Demo" )
   oWV:SetHtml( cHtml )
   oWV:bOnBind := { |hParams, oWV| MsgInfo( hParams[ "msg" ] ) }
   oWV:Center()
   oWV:Run()
   oWV:End()

return nil

Exemplo: Chamar JavaScript a partir do Harbour

#include "FiveWin.ch"

function Main()

   local oWnd, oWV, oBtn

   DEFINE WINDOW oWnd TITLE "WebView2 + Controls"

   oWV := TWebView2():New( oWnd )
   oWV:Navigate( "https://www.example.com" )

   // Após a navegação ser concluída, injetar JavaScript
   oWV:bOnNavigationCompleted := { |cUrl|
      // Mudar o fundo da página
      oWV:Eval( "document.body.style.backgroundColor = '#f0f0f0'" )

      // Obter o título da página
      oWV:Eval( "document.title" )
   }

   // Receber o resultado do Eval
   oWV:bOnEval := { |cJson| oWnd:SetText( "Page: " + cJson ) }

   ACTIVATE WINDOW oWnd MAXIMIZED

   oWV:End()

return nil

Exemplo: Incorporar em um Diálogo

#include "FiveWin.ch"

function Main()

   local oDlg, oWV

   DEFINE DIALOG oDlg TITLE "Embedded WebView" SIZE 800, 600

   ACTIVATE DIALOG oDlg ;
      ON INIT ( ;
         oWV := TWebView2():New( oDlg ),  ;
         oWV:Navigate( "https://www.fivetechsoft.com" ) ;
      ) ;
      VALID ( oWV:End(), .T. )

return nil

Função Auxiliar: FW_WebView()

// Uma única linha para exibir HTML em um WebView
FW_WebView( cHtml, oWnd )

FW_WebView() é uma função de conveniência que cria ou reutiliza um TWebView2 para exibir conteúdo HTML. Se oWnd for um TWebView2 existente, ele o reutiliza; caso contrário, cria um novo.

Requisitos

  • Windows 10 ou posterior com o Runtime WebView2 instalado (incluído no Windows 11 por padrão)
  • O Runtime WebView2 é um download separado para Windows 10: Runtime WebView2
  • Os dados do usuário (cookies, cache) são armazenados em cUserDataFolder. Se não especificado, um local padrão é usado.
