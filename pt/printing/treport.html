TReport

Fonte: source/classes/report.prg

TReport é o motor de relatórios de alto nível do FiveWin. Ele automatiza a criação de relatórios tabulares a partir de arquivos DBF ou qualquer fonte de dados, com suporte para cabeçalhos de coluna, agrupamento, subtotais, totais gerais, numeração de páginas, sombras, linhas de grade e exportação para Excel. TReport usa TPrinter internamente para a saída.

Fluxo de Processamento do Relatório

flowchart TD A[Início] --> B[Bloco bInit] B --> C[IniciarPágina] C --> D{Verificar Grupos} D -->|Novo Grupo| E[IniciarGrupo
Imprimir Cabeçalho do Grupo] D -->|Mesmo Grupo| F[Imprimir Dados da Coluna] E --> F F --> G[FimLinha / Pular] G --> H{Fim dos Dados?} H -->|Não| I{Página Cheia?} I -->|Sim| J[FimPágina / IniciarPágina] I -->|Não| D J --> D H -->|Sim| K[FimGrupo
Imprimir Totais] K --> L[Linha de Total Geral] L --> M[FimPágina] M --> N[Bloco bEnd] N --> O[Visualizar / Imprimir]

Exemplo: Relatório de Lista Simples

#include "FiveWin.ch"

function Main()

   local oRpt, oFont1, oFont2

   USE customers

   DEFINE FONT oFont1 NAME "Arial" SIZE 0, -14
   DEFINE FONT oFont2 NAME "Arial" SIZE 0, -10

   REPORT oRpt TITLE "Customer List" ;
      FONT oFont1, oFont2 PREVIEW

   COLUMN TITLE "Name"    DATA customers->name    SIZE 30
   COLUMN TITLE "City"    DATA customers->city    SIZE 20
   COLUMN TITLE "State"   DATA customers->state   SIZE  5
   COLUMN TITLE "Phone"   DATA customers->phone   SIZE 15
   COLUMN TITLE "Balance" DATA customers->balance SIZE 12 ;
      PICTURE "999,999.99" TOTAL

   END REPORT

   oFont1:End()
   oFont2:End()

   USE

return nil

Exemplo: Relatório Agrupado com Subtotais

#include "FiveWin.ch"

function Main()

   local oRpt, oFont1, oFont2, oFont3

   USE invoices INDEX invoices

   SET ORDER TO TAG "CUSTOMER"

   DEFINE FONT oFont1 NAME "Arial Bold" SIZE 0, -16
   DEFINE FONT oFont2 NAME "Arial"      SIZE 0, -10
   DEFINE FONT oFont3 NAME "Arial Bold" SIZE 0, -11

   REPORT oRpt ;
      TITLE "Invoice Report by Customer" ;
      HEADER "Period: " + DToC( Date() - 365 ) + " to " + DToC( Date() ) ;
      FOOTER "Page: " + Str( oRpt:nPage, 3 ) ;
      FONT oFont1, oFont2, oFont3 ;
      PREVIEW

   // Agrupar por cliente
   GROUP ON invoices->custid ;
      HEADER "Customer: " + invoices->custname ;
      FOOTER "Subtotal" ;
      FONT 3 ;
      EJECT

   COLUMN TITLE "Invoice #"  DATA invoices->invnum    SIZE 12
   COLUMN TITLE "Date"       DATA invoices->invdate   SIZE 10
   COLUMN TITLE "Description" DATA invoices->descript SIZE 30
   COLUMN TITLE "Amount"     DATA invoices->amount    SIZE 12 ;
      PICTURE "999,999.99" TOTAL

   END REPORT

   oFont1:End()
   oFont2:End()
   oFont3:End()

   USE

return nil

Exemplo: Relatório Personalizado com Eventos

#include "FiveWin.ch"

function Main()

   local oRpt, oFont1, oFont2
   local nTotal := 0

   USE products INDEX products

   DEFINE FONT oFont1 NAME "Calibri" SIZE 0, -14
   DEFINE FONT oFont2 NAME "Calibri" SIZE 0, -10

   REPORT oRpt TITLE "Product Catalog" ;
      FONT oFont1, oFont2 PREVIEW

   COLUMN TITLE "Code"     DATA products->code     SIZE 10
   COLUMN TITLE "Product"  DATA products->name     SIZE 30
   COLUMN TITLE "Category" DATA products->category SIZE 15
   COLUMN TITLE "Price"    DATA products->price    SIZE 12 ;
      PICTURE "999,999.99" TOTAL
   COLUMN TITLE "Stock"    DATA products->stock    SIZE  8 TOTAL

   END REPORT

   // Ativação personalizada com blocos de eventos
   oRpt:Activate( ;
      { || products->active },                ; // bFor: apenas produtos ativos
      ,                                        ; // bWhile
      { || oRpt:Margin( 1, 1, 1 ) },         ; // bInit
      { || MsgInfo( "Report completed" ) },   ; // bEnd
      { || oRpt:Say( 1, "Page " + ;
         LTrim( Str( oRpt:nPage ) ), 2 ) },  ; // bStartPage
      ,                                        ; // bEndPage
      ,                                        ; // bStartGroup
      )                                          // bEndGroup

   oFont1:End()
   oFont2:End()

   USE

return nil

Exemplo: Exportar para Excel

#include "FiveWin.ch"

function Main()

   local oRpt

   USE customers

   REPORT oRpt TITLE "Customers" PREVIEW

   COLUMN TITLE "Name"    DATA customers->name    SIZE 30
   COLUMN TITLE "Email"   DATA customers->email   SIZE 30
   COLUMN TITLE "Balance" DATA customers->balance SIZE 12 ;
      PICTURE "999,999.99" TOTAL

   END REPORT

   // Exportar para Excel após a visualização
   oRpt:ToExcel( { |n| SysRefresh() } )

   USE

return nil

Dicas

  • Os índices de fontes nos comandos COLUMN e GROUP são base 1 e referem-se ao array FONT no comando REPORT.
  • Use EJECT em GROUP para forçar uma quebra de página quando o grupo muda.
  • O modo lSummary imprime apenas cabeçalhos/rodapés de grupo e totais -- sem linhas de detalhe.
  • O método CellView() habilita uma aparência tipo planilha com bordas de grade ao redor de cada célula.
  • Use nRptAlign para controlar como o relatório é posicionado na página (RPT_LEFT, RPT_CENTER, RPT_RIGHT).
  • NeedNewPage() retorna .T. quando a posição da linha atual atingiu a margem inferior, útil em bStartLine para inserir quebras de página manuais.
