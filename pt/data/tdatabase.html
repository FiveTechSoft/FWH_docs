TDatabase

Source: source/classes/database.prg

TDatabase e o wrapper orientado a objetos do FiveWin em torno do sistema DBF/RDD (Replaceable Database Driver) do Harbour. Encapsula todas as operacoes comuns de banco de dados -- abertura, navegacao, edicao, filtragem, indexacao e mais -- em uma interface de classe limpa. TDatabase pode trabalhar com qualquer RDD suportado pelo Harbour, incluindo DBFCDX, DBFNTX, DBFNSX e ADS (Advantage Database Server).

Construtores

New()

METHOD New( ncArea, cFile, cDriver, lShared, lReadOnly, cPassword )

Cria um objeto TDatabase. Se ncArea for um numero ou alias de uma area de trabalho ja aberta, o TDatabase encapsula a area existente. Se ncArea for uma string de caminho de arquivo, prepara para abrir esse arquivo. Se apenas um parametro for passado, detecta automaticamente se e um numero de area, alias ou nome de arquivo.

Open()

METHOD Open( cAlias, cFile, cDriver, lShared, lReadOnly, cPassword )

Abre um arquivo DBF com alias explicito. Este e o construtor preferido para abrir novos arquivos.

Create()

METHOD Create( cFile, aStruct, cDriver, cIndexFields, cPassword )

Cria um novo arquivo DBF com a estrutura especificada e opcionalmente cria um indice.

Exemplo: Operacoes Basicas com DBF

#include "FiveWin.ch"

function Main()

   local oDb, oWnd

   // Open a DBF file with DBFCDX driver
   DATABASE oDb FILE "customers.dbf" DRIVER "DBFCDX" SHARED

   // Equivalent to:
   // oDb := TDatabase():Open( "CUST", "customers.dbf", "DBFCDX", .T. )

   // Navigate
   oDb:GoTop()
   ? oDb:FieldGet( "FIRST" )    // "John"
   ? oDb:FieldGet( "LAST" )     // "Smith"
   ? oDb:hRec[ "FIRST" ]        // "John" (hash access)

   // Set an index order and seek
   oDb:SetOrder( "LAST" )
   if oDb:Seek( "Jones" )
      ? "Found:", oDb:FieldGet( "FIRST" ), oDb:FieldGet( "LAST" )
   endif

   // Edit a record
   if oDb:RecLock()
      oDb:FieldPut( "PHONE", "555-1234" )
      oDb:UnLock()
   endif

   // Append a new record
   oDb:Append()
   oDb:FieldPut( "FIRST", "Jane" )
   oDb:FieldPut( "LAST",  "Doe" )
   oDb:FieldPut( "PHONE", "555-5678" )
   oDb:Save()

   // Apply a filter
   oDb:SetFilter( "STATE = 'CA'" )
   oDb:GoTop()
   do while ! oDb:Eof()
      ? oDb:FieldGet( "FIRST" ), oDb:FieldGet( "LAST" )
      oDb:Skip()
   enddo
   oDb:ClearFilter()

   oDb:Close()

return nil

Exemplo: Browse com XBrowse

#include "FiveWin.ch"

function Main()

   local oDb, oWnd, oBrw

   DATABASE oDb FILE "customers.dbf" DRIVER "DBFCDX" SHARED

   DEFINE WINDOW oWnd TITLE "Customer Browser"

   oDb:xBrowse( oWnd,, .T., { |oBrw|
      oBrw:cHeaders := { "First Name", "Last Name", "Phone", "State" }
      oBrw:lRecordSelector := .F.
   } )

   ACTIVATE WINDOW oWnd MAXIMIZED

   oDb:Close()

return nil

Exemplo: Edicao com Buffer

#include "FiveWin.ch"

function Main()

   local oDb

   DATABASE oDb FILE "customers.dbf" DRIVER "DBFCDX" SHARED

   // Enable buffering for optimistic locking
   oDb:SetBuffer( .T. )

   oDb:GoTop()

   // Changes go to buffer, not disk
   oDb:FieldPut( "PHONE", "555-0000" )

   if MsgYesNo( "Save changes?" )
      oDb:Save()        // writes buffer to disk
   else
      oDb:RollBack()    // discard changes
   endif

   oDb:Close()

return nil

Exemplo: Criando Indices

#include "FiveWin.ch"

function Main()

   local oDb

   DATABASE oDb FILE "customers.dbf" DRIVER "DBFCDX" SHARED

   // Create a compound index with multiple tags
   oDb:CreateIndex( "customers.cdx", "NAME",  "UPPER(LAST)+UPPER(FIRST)" )
   oDb:CreateIndex( "customers.cdx", "STATE", "STATE" )
   oDb:CreateIndex( "customers.cdx", "PHONE", "PHONE", .T. )  // unique

   // Use the NAME order
   oDb:SetOrder( "NAME" )
   oDb:GoTop()

   // Scoped navigation: only records where STATE = "CA"
   oDb:SetScope( "CA", "CA" )
   oDb:GoTop()
   do while ! oDb:Eof()
      ? oDb:FieldGet( "LAST" )
      oDb:Skip()
   enddo

   oDb:Close()

return nil
