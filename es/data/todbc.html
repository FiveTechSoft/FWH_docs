TODBC

Fuente: source/classes/todbc.prg

TODBC proporciona acceso directo a ODBC (Open Database Connectivity) desde aplicaciones Harbour. A diferencia de TRecSet que depende de ADO/COM, TODBC llama a las funciones de la API ODBC de Windows directamente a través de la interfaz a nivel C de Harbour. Esto lo hace ligero y adecuado para conectarse a cualquier fuente de datos compatible con ODBC incluyendo SQL Server, MySQL, PostgreSQL, Oracle, SQLite y más.

Métodos

Constructor

METHOD New( cDSN, cUser, cPassword )

Crea una nueva conexión ODBC. El parámetro cDSN puede ser un nombre de DSN registrado en el Administrador de Orígenes de Datos ODBC, o una cadena de conexión completa que comience por "DRIVER=". El constructor reserva los handles de entorno y de conexión ODBC e intenta conectar de inmediato. Compruebe oOdbc:lSuccess tras la creación.

Ejemplo: Conectar vía DSN

#include "FiveWin.ch"

function Main()

   local oOdbc, aResult, aFields, aData

   // Conectar usando un DSN preconfigurado
   oOdbc := TOdbc():New( "MyDSN", "myuser", "mypass" )

   if ! oOdbc:lSuccess
      MsgStop( "Connection failed: " + oOdbc:LastError() )
      return nil
   endif

   ? "Connected to:", oOdbc:cDSN

   // Listar las tablas
   ? "Tables:", oOdbc:GetTables()

   // Ejecutar una consulta SELECT
   aResult := oOdbc:Query( "SELECT * FROM Customers WHERE Country = 'USA'" )

   if ! Empty( aResult )
      aFields := aResult[ 1 ]   // descriptores de campo
      aData   := aResult[ 2 ]   // filas de datos
      ? "Found", Len( aData ), "records"
      ? "Fields:", Len( aFields )
   endif

   oOdbc:End()

return nil

Ejemplo: Conectar vía Cadena de Driver

#include "FiveWin.ch"

function Main()

   local oOdbc

   // Conectar a SQL Server sin un DSN
   oOdbc := TOdbc():New( ;
      "DRIVER={SQL Server};SERVER=localhost\SQLEXPRESS;" + ;
      "DATABASE=Northwind;Trusted_Connection=yes" )

   if oOdbc:lSuccess
      // Ejecutar un comando
      oOdbc:Execute( "UPDATE Products SET UnitPrice = UnitPrice * 1.10 " + ;
                     "WHERE CategoryID = 1" )
      oOdbc:Commit()
      ? "Prices updated"
   endif

   oOdbc:End()

return nil

Ejemplo: Obtener Fila por Fila

#include "FiveWin.ch"
#include "sql.ch"

function Main()

   local oOdbc, hStmt, aFields, n

   oOdbc := TOdbc():New( "MyDSN", "user", "pass" )

   if ! oOdbc:lSuccess
      return nil
   endif

   // Ejecutar y obtener el handle de la sentencia
   hStmt := oOdbc:Execute( "SELECT Name, Email, Balance FROM Clients" )

   if hStmt != 0
      // Obtener la información de los campos
      aFields := oOdbc:CursorFields( hStmt )

      // Obtener fila por fila
      do while oOdbc:Fetch( hStmt )
         for n := 1 to Len( aFields )
            ?? oOdbc:GetValue( n, hStmt, aFields ), " | "
         next
         ?   // nueva línea
      enddo

      oOdbc:DropStatement( hStmt )
   endif

   oOdbc:End()

return nil

Ejemplo: Transacciones

#include "FiveWin.ch"

function Main()

   local oOdbc

   oOdbc := TOdbc():New( "MyDSN", "user", "pass" )

   if oOdbc:lSuccess
      // Desactivar el auto-commit para el control manual de transacciones
      // (el auto-commit está activado por defecto tras New())

      oOdbc:Execute( "INSERT INTO Orders (CustomerID, OrderDate) VALUES (1, '2026-03-24')" )
      oOdbc:Execute( "INSERT INTO OrderItems (OrderID, ProductID, Qty) VALUES (SCOPE_IDENTITY(), 42, 5)" )

      if oOdbc:IsError()
         oOdbc:RollBack()
         MsgStop( "Transaction failed" )
      else
         oOdbc:Commit()
         MsgInfo( "Order created" )
      endif
   endif

   oOdbc:End()

return nil

Cadenas de Conexión Comunes

// SQL Server (DSN-less)
"DRIVER={SQL Server};SERVER=myserver;DATABASE=mydb;UID=user;PWD=pass"

// MySQL vía ODBC
"DRIVER={MySQL ODBC 8.0 Unicode Driver};SERVER=localhost;DATABASE=mydb;USER=root;PASSWORD=pass;PORT=3306"

// PostgreSQL
"DRIVER={PostgreSQL Unicode};SERVER=localhost;DATABASE=mydb;UID=user;PWD=pass;PORT=5432"

// SQLite
"DRIVER={SQLite3 ODBC Driver};Database=C:\data\mydb.sqlite"

// MS Access
"DRIVER={Microsoft Access Driver (*.mdb, *.accdb)};DBQ=C:\data\mydb.accdb"
