TWebView2

Fuente: source/classes/twebview2.prg

TWebView2 integra el motor del navegador Microsoft Edge (Chromium) en aplicaciones FiveWin. Proporciona un motor de renderizado moderno HTML5/CSS3/JavaScript que puede navegar URLs, mostrar HTML local, ejecutar JavaScript desde Harbour y recibir callbacks de JavaScript de vuelta al código Harbour. Esto reemplaza el antiguo enfoque basado en Internet Explorer TActiveX("Shell.Explorer.2").

Ejemplo: Navegar a una URL

#include "FiveWin.ch"

function Main()

   local oWV

   oWV := TWebView2():New()
   oWV:SetTitle( "FiveWin Browser" )
   oWV:Navigate( "https://www.fivetechsoft.com" )
   oWV:Center()
   oWV:Run()
   oWV:End()

return nil

Ejemplo: Mostrar HTML Local

#include "FiveWin.ch"

function Main()

   local oWV, cHtml

   TEXT INTO cHtml
   <!DOCTYPE html>
   <html>
   <head>
      <style>
         body { font-family: Segoe UI; margin: 40px; background: #1a1a2e; color: #eee; }
         h1 { color: #e94560; }
         .card { background: #16213e; padding: 20px; border-radius: 10px; margin: 20px 0; }
         button { background: #e94560; color: white; border: none; padding: 10px 20px;
                  border-radius: 5px; cursor: pointer; font-size: 14px; }
         button:hover { background: #c73a52; }
      </style>
   </head>
   <body>
      <h1>FiveWin + WebView2</h1>
      <div class="card">
         <p>Este HTML es renderizado por el motor Chromium dentro de una ventana FiveWin.</p>
         <button onclick="SendToFWH({action:'hello', msg:'Button clicked!'})">
            Haz Clic
         </button>
      </div>
   </body>
   </html>
   ENDTEXT

   oWV := TWebView2():New()
   oWV:SetTitle( "HTML Demo" )
   oWV:SetHtml( cHtml )
   oWV:bOnBind := { |hParams, oWV| MsgInfo( hParams[ "msg" ] ) }
   oWV:Center()
   oWV:Run()
   oWV:End()

return nil

Ejemplo: Llamar JavaScript desde Harbour

#include "FiveWin.ch"

function Main()

   local oWnd, oWV, oBtn

   DEFINE WINDOW oWnd TITLE "WebView2 + Controls"

   oWV := TWebView2():New( oWnd )
   oWV:Navigate( "https://www.example.com" )

   // Después de que la navegación se complete, inyectar JavaScript
   oWV:bOnNavigationCompleted := { |cUrl|
      // Cambiar el fondo de la página
      oWV:Eval( "document.body.style.backgroundColor = '#f0f0f0'" )

      // Obtener el título de la página
      oWV:Eval( "document.title" )
   }

   // Recibir el resultado de Eval
   oWV:bOnEval := { |cJson| oWnd:SetText( "Page: " + cJson ) }

   ACTIVATE WINDOW oWnd MAXIMIZED

   oWV:End()

return nil

Ejemplo: Integrar en un Diálogo

#include "FiveWin.ch"

function Main()

   local oDlg, oWV

   DEFINE DIALOG oDlg TITLE "Embedded WebView" SIZE 800, 600

   ACTIVATE DIALOG oDlg ;
      ON INIT ( ;
         oWV := TWebView2():New( oDlg ),  ;
         oWV:Navigate( "https://www.fivetechsoft.com" ) ;
      ) ;
      VALID ( oWV:End(), .T. )

return nil

Función Auxiliar: FW_WebView()

// Una sola línea para mostrar HTML en un WebView
FW_WebView( cHtml, oWnd )

FW_WebView() es una función de conveniencia que crea o reutiliza un TWebView2 para mostrar contenido HTML. Si oWnd es un TWebView2 existente, lo reutiliza; de lo contrario crea uno nuevo.

Requisitos

  • Windows 10 o posterior con el Runtime WebView2 instalado (incluido con Windows 11 por defecto)
  • El Runtime WebView2 es una descarga separada para Windows 10: Runtime WebView2
  • Los datos de usuario (cookies, caché) se almacenan en cUserDataFolder. Si no se especifica, se usa una ubicación predeterminada.
