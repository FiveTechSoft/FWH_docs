TWebServer / TSocket

Fuente: source/classes/twebserv.prg (TWebServer), source/classes/tsocket.prg (TSocket)

TWebServer es el servidor HTTP integrado de FiveWin que puede servir páginas HTML, procesar solicitudes GET/POST y combinar contenido dinámico en plantillas HTML. Está construido sobre TSocket, que proporciona comunicación de sockets TCP/IP de bajo nivel usando la API Winsock de Windows.

Ejemplo: Servidor HTTP Básico

#include "FiveWin.ch"

function Main()

   local oWnd, oWS

   DEFINE WINDOW oWnd TITLE "FiveWin Web Server"

   // Crear servidor web en el puerto 8080
   oWS := TWebServer():New( 8080 )
   oWS:cDefPath := "www"
   oWS:cDefPage := "index.html"
   oWS:Activate()

   // Abrir navegador
   oWS:Explore( "index.html" )

   ACTIVATE WINDOW oWnd ;
      VALID ( oWS:End(), .T. )

return nil

Ejemplo: Contenido Dinámico

#include "FiveWin.ch"

function Main()

   local oWnd, oWS

   DEFINE WINDOW oWnd TITLE "Dynamic Web Server"

   oWS := TWebServer():New( 8080 )
   oWS:lDebug := .T.
   oWS:cLogFile := "webserver.log"
   oWS:Activate()

   oWS:Explore()

   ACTIVATE WINDOW oWnd VALID ( oWS:End(), .T. )

return nil

Ejemplo: Cliente de Socket TCP Sin Procesar

#include "FiveWin.ch"

function Main()

   local oSocket, cResponse

   oSocket := TSocket():New( 0 )

   // Conectar a un servidor remoto
   oSocket:Connect( "93.184.216.34", 80 )  // example.com

   // Enviar solicitud HTTP
   oSocket:SendData( ;
      "GET / HTTP/1.1" + CRLF + ;
      "Host: example.com" + CRLF + ;
      "Connection: close" + CRLF + CRLF )

   // Leer respuesta
   oSocket:bRead := { |oSock|
      cResponse := oSock:GetData()
      ? cResponse
   }

   oSocket:bClose := { |oSock|
      ? "Conexión cerrada"
      oSock:End()
   }

   // Procesar eventos (en una app real, el bucle de mensajes lo maneja)
   SysRefresh()
   Inkey( 5 )

   oSocket:End()

return nil

Ejemplo: Servidor de Chat Simple

#include "FiveWin.ch"

static aClients := {}

function Main()

   local oWnd, oSocket

   DEFINE WINDOW oWnd TITLE "Chat Server - Port 9000"

   oSocket := TSocket():New( 9000 )

   oSocket:bAccept := { ||
      local oClient := TSocket():Accept( oSocket:nSocket )
      AAdd( aClients, oClient )

      oClient:bRead := { |oSock|
         local cMsg := oSock:GetData()
         // Difundir a todos los clientes
         AEval( aClients, { |o| o:SendData( cMsg ) } )
      }

      oClient:bClose := { |oSock|
         local nAt := AScan( aClients, { |o| o:nSocket == oSock:nSocket } )
         if nAt > 0
            ADel( aClients, nAt, .T. )
         endif
         oSock:End()
      }
   }

   oSocket:Listen()

   ACTIVATE WINDOW oWnd VALID ( oSocket:End(), .T. )

return nil

Notas

  • TWebServer crea un directorio www automáticamente si no existe, con una página HTML predeterminada.
  • El método HtmlMerge() procesa etiquetas <%=expresión%> en archivos HTML, evaluando expresiones Harbour e incrustando los resultados.
  • Los datos de cookies de las solicitudes HTTP se analizan automáticamente y se almacenan en oWS:cCookie.
  • TSocket usa eventos de socket asíncronos de Windows (FD_READ, FD_WRITE, FD_ACCEPT, FD_CLOSE) a través del bucle de mensajes, por lo que funciona sin problemas con la arquitectura orientada a eventos de FiveWin.
  • Para uso en producción, considere usar un proxy inverso (nginx, Apache) delante de TWebServer para terminación SSL y servir archivos estáticos.
