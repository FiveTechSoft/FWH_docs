TReport

Fuente: source/classes/report.prg

TReport es el motor de informes de alto nivel de FiveWin. Automatiza la creación de informes tabulares a partir de archivos DBF o cualquier fuente de datos, con soporte para encabezados de columna, agrupación, subtotales, totales generales, numeración de páginas, sombras, líneas de cuadrícula y exportación a Excel. TReport utiliza TPrinter internamente para la salida.

Flujo de Procesamiento del Informe

flowchart TD A[Inicio] --> B[Bloque bInit] B --> C[IniciarPágina] C --> D{Verificar Grupos} D -->|Nuevo Grupo| E[IniciarGrupo
Imprimir Encabezado de Grupo] D -->|Mismo Grupo| F[Imprimir Datos de Columna] E --> F F --> G[FinLínea / Saltar] G --> H{¿Fin de Datos?} H -->|No| I{¿Página Llena?} I -->|Sí| J[FinPágina / IniciarPágina] I -->|No| D J --> D H -->|Sí| K[FinGrupo
Imprimir Totales] K --> L[Línea de Total General] L --> M[FinPágina] M --> N[Bloque bEnd] N --> O[Vista Previa / Imprimir]

Ejemplo: Informe de Lista Simple

#include "FiveWin.ch"

function Main()

   local oRpt, oFont1, oFont2

   USE customers

   DEFINE FONT oFont1 NAME "Arial" SIZE 0, -14
   DEFINE FONT oFont2 NAME "Arial" SIZE 0, -10

   REPORT oRpt TITLE "Customer List" ;
      FONT oFont1, oFont2 PREVIEW

   COLUMN TITLE "Name"    DATA customers->name    SIZE 30
   COLUMN TITLE "City"    DATA customers->city    SIZE 20
   COLUMN TITLE "State"   DATA customers->state   SIZE  5
   COLUMN TITLE "Phone"   DATA customers->phone   SIZE 15
   COLUMN TITLE "Balance" DATA customers->balance SIZE 12 ;
      PICTURE "999,999.99" TOTAL

   END REPORT

   oFont1:End()
   oFont2:End()

   USE

return nil

Ejemplo: Informe Agrupado con Subtotales

#include "FiveWin.ch"

function Main()

   local oRpt, oFont1, oFont2, oFont3

   USE invoices INDEX invoices

   SET ORDER TO TAG "CUSTOMER"

   DEFINE FONT oFont1 NAME "Arial Bold" SIZE 0, -16
   DEFINE FONT oFont2 NAME "Arial"      SIZE 0, -10
   DEFINE FONT oFont3 NAME "Arial Bold" SIZE 0, -11

   REPORT oRpt ;
      TITLE "Invoice Report by Customer" ;
      HEADER "Period: " + DToC( Date() - 365 ) + " to " + DToC( Date() ) ;
      FOOTER "Page: " + Str( oRpt:nPage, 3 ) ;
      FONT oFont1, oFont2, oFont3 ;
      PREVIEW

   // Agrupar por cliente
   GROUP ON invoices->custid ;
      HEADER "Customer: " + invoices->custname ;
      FOOTER "Subtotal" ;
      FONT 3 ;
      EJECT

   COLUMN TITLE "Invoice #"  DATA invoices->invnum    SIZE 12
   COLUMN TITLE "Date"       DATA invoices->invdate   SIZE 10
   COLUMN TITLE "Description" DATA invoices->descript SIZE 30
   COLUMN TITLE "Amount"     DATA invoices->amount    SIZE 12 ;
      PICTURE "999,999.99" TOTAL

   END REPORT

   oFont1:End()
   oFont2:End()
   oFont3:End()

   USE

return nil

Ejemplo: Informe Personalizado con Eventos

#include "FiveWin.ch"

function Main()

   local oRpt, oFont1, oFont2
   local nTotal := 0

   USE products INDEX products

   DEFINE FONT oFont1 NAME "Calibri" SIZE 0, -14
   DEFINE FONT oFont2 NAME "Calibri" SIZE 0, -10

   REPORT oRpt TITLE "Product Catalog" ;
      FONT oFont1, oFont2 PREVIEW

   COLUMN TITLE "Code"     DATA products->code     SIZE 10
   COLUMN TITLE "Product"  DATA products->name     SIZE 30
   COLUMN TITLE "Category" DATA products->category SIZE 15
   COLUMN TITLE "Price"    DATA products->price    SIZE 12 ;
      PICTURE "999,999.99" TOTAL
   COLUMN TITLE "Stock"    DATA products->stock    SIZE  8 TOTAL

   END REPORT

   // Activación personalizada con bloques de eventos
   oRpt:Activate( ;
      { || products->active },                ; // bFor: solo productos activos
      ,                                        ; // bWhile
      { || oRpt:Margin( 1, 1, 1 ) },         ; // bInit
      { || MsgInfo( "Report completed" ) },   ; // bEnd
      { || oRpt:Say( 1, "Page " + ;
         LTrim( Str( oRpt:nPage ) ), 2 ) },  ; // bStartPage
      ,                                        ; // bEndPage
      ,                                        ; // bStartGroup
      )                                          // bEndGroup

   oFont1:End()
   oFont2:End()

   USE

return nil

Ejemplo: Exportar a Excel

#include "FiveWin.ch"

function Main()

   local oRpt

   USE customers

   REPORT oRpt TITLE "Customers" PREVIEW

   COLUMN TITLE "Name"    DATA customers->name    SIZE 30
   COLUMN TITLE "Email"   DATA customers->email   SIZE 30
   COLUMN TITLE "Balance" DATA customers->balance SIZE 12 ;
      PICTURE "999,999.99" TOTAL

   END REPORT

   // Exportar a Excel después de la vista previa
   oRpt:ToExcel( { |n| SysRefresh() } )

   USE

return nil

Consejos

  • Los índices de fuentes en los comandos COLUMN y GROUP son base 1 y se refieren al array FONT del comando REPORT.
  • Use EJECT en GROUP para forzar un salto de página cuando cambia el grupo.
  • El modo lSummary imprime solo encabezados/pies de grupo y totales -- sin líneas de detalle.
  • El método CellView() habilita una apariencia tipo hoja de cálculo con bordes de cuadrícula alrededor de cada celda.
  • Use nRptAlign para controlar cómo se posiciona el informe en la página (RPT_LEFT, RPT_CENTER, RPT_RIGHT).
  • NeedNewPage() devuelve .T. cuando la posición de fila actual ha alcanzado el margen inferior, útil en bStartLine para insertar saltos de página manuales.
