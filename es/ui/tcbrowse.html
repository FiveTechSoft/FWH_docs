TCBrowse

Fuente: source/classes/tcbrowse.prg

Hereda de: TWBrowse

TCBrowse is a column-based browse control that extends TWBrowse with multi-column support, column drag-reorder, frozen columns, incremental seek, inline editing, bitmap columns, and built-in report generation. It can be bound to both arrays and DBF objects, making it a versatile data-browsing control for tabular data.

Ejemplo: Explorador de 3 Columnas

#include "FiveWin.ch"

function Main()

   local oWnd, oBrowse, oCol1, oCol2, oCol3
   local aData := { { "Widget A", 150, Date() },;
                    { "Widget B", 275, Date() - 1 },;
                    { "Widget C", 100, Date() - 2 } }

   DEFINE WINDOW oWnd TITLE "TCBrowse Demo" SIZE 600, 400

   @ 10, 10 BROWSE oBrowse ;
      SIZE 580, 360 ;
      OF oWnd

   oBrowse:SetArray( aData )

   oCol1 := TCColumn():New( "Item",   { || aData[ oBrowse:nArrayAt, 1 ] } )
   oCol2 := TCColumn():New( "Price",  { || aData[ oBrowse:nArrayAt, 2 ] }, "999.99" )
   oCol3 := TCColumn():New( "Date",   { || aData[ oBrowse:nArrayAt, 3 ] }, "D" )

   oBrowse:AddColumn( oCol1 )
   oBrowse:AddColumn( oCol2 )
   oBrowse:AddColumn( oCol3 )

   ACTIVATE WINDOW oWnd CENTERED

return nil
