function Main()

   local oWnd, oBmp

   DEFINE WINDOW oWnd TITLE "TBitmap Demo" SIZE 600, 450

   @ 10, 10 BITMAP oBmp OF oWnd ;
      FILE "c:\fwh\bitmaps\olga1.bmp" ;
      SIZE 300, 200

   ACTIVATE WINDOW oWnd CENTERED
